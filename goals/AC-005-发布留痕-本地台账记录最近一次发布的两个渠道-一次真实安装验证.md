criterion: >-
  CAR=".quay/release-runs.jsonl"

  if [ ! -f "$CAR" ]; then
    echo "CAUSE=carrier-absent — 没有发布台账 $CAR；「没查成」不等于「合格」" >&2; exit 1
  fi

  python3 - <<'PY'

  import json, sys

  rows = [json.loads(l) for l in open('.quay/release-runs.jsonl') if l.strip()]

  if not rows:
      sys.stderr.write('CAUSE=carrier-empty — 台账存在但没有任何发布记录\n'); sys.exit(1)
  last = rows[-1]

  for k in ('cli-publish', 'plugin-publish', 'install-verify'):
      if last.get(k) != 'success':
          sys.stderr.write('CAUSE=release-run-incomplete — 最近一次发布（tag=%s）的 %s 不是 success：%s\n' % (last.get('tag'), k, last)); sys.exit(1)
  sys.exit(0)

  PY
expect: 本地存在 .quay/release-runs.jsonl，且其最后一行同时记录
  cli-publish=success、plugin-publish=success、install-verify=success。
