  import json, sys

  core = json.load(open('package.json'))['version']

  mk   = json.load(open('.claude-plugin/marketplace.json'))

  pp   = json.load(open('plugin/package.json'))

  pj   = json.load(open('plugin/.claude-plugin/plugin.json'))

  got = {
      'package.json': core,
      'package-lock.json': lock['version'],
      'package-lock.json#packages[""]': lock['packages']['']['version'],
      '.claude-plugin/marketplace.json (source pin)': mk['plugins'][0]['source']['version'],
      'plugin/package.json': pp['version'],
      'plugin/.claude-plugin/plugin.json': pj['version'],
  }

  bad = {k: v for k, v in got.items() if v != core}

  if bad:
      sys.stderr.write('CAUSE=version-carriers-disagree — 期望全部等于 %s，实际不符：%s\n' % (core, bad)); sys.exit(1)
  print('all carriers == %s' % core)

  sys.exit(0)

  PY
